window.generarPDFCaducidades = function() { var items = window.datos.caducidades || []; if (items.length === 0) { window.mostrarToast('⚠️ No hay productos', '⚠️'); return; } try { // Verificar que jsPDF esté disponible if (typeof window.jspdf === 'undefined' || typeof window.jspdf.jsPDF === 'undefined') { window.mostrarToast('❌ Librería PDF no disponible. Recarga la página.', '❌'); return; } var { jsPDF } = window.jspdf; var doc = new jsPDF('p', 'pt', 'a4'); // ========== ENCABEZADO ========== doc.setFontSize(18); doc.setTextColor(255, 165, 0); // Naranja doc.text('📋 CADUCIDADES NETO', 40, 50); doc.setFontSize(10); doc.setTextColor(100, 100, 100); // Gris oscuro doc.text('Fecha de generación: ' + new Date().toLocaleDateString(), 40, 72); doc.text('Hora: ' + new Date().toLocaleTimeString(), 40, 87); doc.text('Usuario: ' + (window.usuarioActual ? window.usuarioActual.nombre : 'N/A'), 40, 102); doc.text('Total de productos: ' + items.length, 40, 117); doc.text('Tienda: MIGUEL HIDALGO PUEBLA', 40, 132); // Línea separadora doc.setDrawColor(255, 165, 0); doc.setLineWidth(1); doc.line(40, 145, 560, 145); // ========== ENCABEZADOS DE TABLA ========== var y = 165; doc.setFontSize(9); doc.setTextColor(255, 165, 0); // Naranja doc.setFont('helvetica', 'bold'); // Anchos de columna var colX = { num: 40, nombre: 70, codigo: 200, fecha: 300, cantidad: 380, costo: 440, total: 510 }; doc.text('#', colX.num, y); doc.text('Producto', colX.nombre, y); doc.text('Código', colX.codigo, y); doc.text('Fecha Cad.', colX.fecha, y); doc.text('Cant.', colX.cantidad, y); doc.text('Costo', colX.costo, y); doc.text('Total', colX.total, y); y += 8; doc.setDrawColor(200, 200, 200); doc.setLineWidth(0.5); doc.line(40, y, 560, y); y += 10; // ========== LISTA DE PRODUCTOS ========== var totalValor = 0; var pageCount = 1; var maxY = 750; doc.setFont('helvetica', 'normal'); doc.setFontSize(8); doc.setTextColor(50, 50, 50); // Gris oscuro para mejor legibilidad items.forEach(function(p, i) { var costoTotal = (p.costo || 0) * (p.cantidad || 0); totalValor += costoTotal; var nombre = (p.nombre || 'Sin nombre').substring(0, 20); var codigo = (p.codigo || '').substring(0, 12); var fecha = p.fecha || 'N/A'; var cantidad = p.cantidad || 0; var costo = (p.costo || 0).toFixed(2); var total = costoTotal.toFixed(2); // Verificar si necesitamos nueva página if (y > maxY) { doc.addPage(); pageCount++; y = 50; // Reimprimir encabezados en nueva página doc.setFontSize(10); doc.setTextColor(255, 165, 0); doc.setFont('helvetica', 'bold'); doc.text('📋 CADUCIDADES NETO (cont.)', 40, y); y += 15; doc.setFontSize(9); doc.text('#', colX.num, y); doc.text('Producto', colX.nombre, y); doc.text('Código', colX.codigo, y); doc.text('Fecha Cad.', colX.fecha, y); doc.text('Cant.', colX.cantidad, y); doc.text('Costo', colX.costo, y); doc.text('Total', colX.total, y); y += 8; doc.line(40, y, 560, y); y += 10; doc.setFont('helvetica', 'normal'); doc.setFontSize(8); doc.setTextColor(50, 50, 50); } // Dibujar fila doc.text((i + 1).toString(), colX.num, y); doc.text(nombre, colX.nombre, y); doc.text(codigo, colX.codigo, y); doc.text(fecha, colX.fecha, y); doc.text(cantidad.toString(), colX.cantidad, y); doc.text('$' + costo, colX.costo, y); doc.text('$' + total, colX.total, y); y += 12; }); // ========== LÍNEA FINAL Y TOTALES ========== y += 5; doc.setDrawColor(255, 165, 0); doc.setLineWidth(1); doc.line(40, y, 560, y); y += 12; doc.setFontSize(12); doc.setTextColor(255, 165, 0); doc.setFont('helvetica', 'bold'); doc.text('💰 VALOR TOTAL: $' + totalValor.toFixed(2), 40, y); y += 20; doc.setFontSize(8); doc.setTextColor(150, 150, 150); doc.setFont('helvetica', 'normal'); doc.text('* Este reporte muestra los productos que están próximos a caducar.', 40, y); y += 12; doc.text('* Revisa periódicamente para evitar pérdidas por caducidad.', 40, y); // ========== PIE DE PÁGINA ========== var totalPaginas = doc.internal.getNumberOfPages(); for (var i = 1; i <= totalPaginas; i++) { doc.setPage(i); doc.setFontSize(8); doc.setTextColor(180, 180, 180); doc.setFont('helvetica', 'italic'); doc.text('NETO Inventario - Página ' + i + ' de ' + totalPaginas, 40, doc.internal.pageSize.height - 20); doc.text('Generado: ' + new Date().toLocaleString(), 40, doc.internal.pageSize.height - 10); } // ========== GUARDAR PDF ========== doc.save('caducidades_neto_' + new Date().toISOString().slice(0,10) + '.pdf'); window.mostrarToast('📄 PDF de caducidades generado correctamente', '📄'); } catch (e) { console.error('Error generando PDF:', e); window.mostrarToast('❌ Error al generar PDF: ' + e.message, '❌'); } };
